local function f1(p1, p2)
  local character = p2.Character

  if not character then
    p2.CharacterAdded:Wait()

  end

  local humanoidRootPart = character:WaitForChild("HumanoidRootPart")
  local humanoid = character:WaitForChild("Humanoid")

  local v2 = false
  local v3 = false
  local walkSpeed = humanoid.WalkSpeed
  local jumpPower = humanoid.JumpPower
  local v4 = { Fly = Enum.KeyCode.F, Speed = Enum.KeyCode.G, Jump = Enum.KeyCode.J }
  local v5 = { Fly = p1.fly.keyLabel, Speed = p1.speed.keyLabel, Jump = p1.jump.keyLabel }

  for key, value in pairs(v5) do
    value.Text = v4[key].Name
  end

  local bodyGyro, bodyVelocity, connect, connect2, connect3

  local function f2()
    if v1 then
      return
    end

    v1 = true
    p1.fly.frame.BackgroundColor3 = Color3.fromRGB(180, 40, 40)
    bodyGyro = Instance.new("BodyGyro")
    bodyVelocity = Instance.new("BodyVelocity")
    bodyGyro.MaxTorque = Vector3.new(100000, 100000, 100000)
    bodyVelocity.MaxForce = Vector3.new(100000, 100000, 100000)
    bodyGyro.P = 90000
    bodyVelocity.P = 90000
    bodyGyro.Parent = humanoidRootPart
    bodyVelocity.Parent = humanoidRootPart
    humanoid.PlatformStand = true

    local v6 = {
      F = 0,
      B = 0,
      L = 0,
      R = 0,
      U = 0,
      D = 0,
    }

    local f3

    function f3(p3, p4)
      local v7 = p4 and 1 or 0

      if p3.KeyCode == Enum.KeyCode.W then
        v6.F = v7
      end

      if p3.KeyCode == Enum.KeyCode.S then
        v6.B = -v7
      end

      if p3.KeyCode == Enum.KeyCode.A then
        v6.L = -v7
      end

      if p3.KeyCode == Enum.KeyCode.D then
        v6.R = v7
      end

      if p3.KeyCode == Enum.KeyCode.Space then
        v6.U = v7
      end

      if p3.KeyCode == Enum.KeyCode.LeftControl then
        v6.D = -v7
      end
    end

    connect2 = userInputService.InputBegan:Connect(function(input) f3(input, true) end)
    connect3 = userInputService.InputEnded:Connect(function(input2) f3(input2, false) end)

    connect = runService.Heartbeat:Connect(function()
      if not v1 then
        return
      else
        bodyGyro.CFrame = workspace.CurrentCamera.CFrame

        local v8 = workspace.CurrentCamera.CFrame.LookVector * (v6.F + v6.B)
          + workspace.CurrentCamera.CFrame.RightVector * (v6.R + v6.L)
          + Vector3.new(0, v6.U + v6.D, 0)

        if v8.Magnitude > 0 then
          bodyVelocity.Velocity = v8.Unit * 80
        else
          bodyVelocity.Velocity = Vector3.zero
        end

        return
      end
    end)
  end

  local function f4()
    if not v1 then
      return
    end

    v1 = false
    p1.fly.frame.BackgroundColor3 = Color3.fromRGB(22, 22, 22)

    if connect then
      connect:Disconnect()
    end

    if connect2 then
      connect2:Disconnect()
    end

    if connect3 then
      connect3:Disconnect()
    end

    if bodyGyro then
      bodyGyro:Destroy()
    end

    if bodyVelocity then
      bodyVelocity:Destroy()
    end

    humanoid.PlatformStand = false
  end

  local function f5()
    v2 = not v2

    local frame = p1.speed.frame
    frame.BackgroundColor3 = v2 and Color3.fromRGB(180, 40, 40) or Color3.fromRGB(22, 22, 22)

    humanoid.WalkSpeed = v2 and 80 or walkSpeed
  end

  local function f6()
    v3 = not v3

    local frame2 = p1.jump.frame
    frame2.BackgroundColor3 = v3 and Color3.fromRGB(180, 40, 40) or Color3.fromRGB(22, 22, 22)

    humanoid.JumpPower = v3 and 150 or jumpPower
  end

  local function f7(parent, p5)
    local textButton = Instance.new("TextButton")
    textButton.Size = UDim2.new(1, 0, 1, 0)
    textButton.BackgroundTransparency = 1
    textButton.Text = ""
    textButton.Parent = parent
    textButton.MouseButton1Click:Connect(p5)
  end

  f7(p1.fly.frame, function()
    if v1 then
      f4()
    else
      f2()
    end
  end)

  f7(p1.speed.frame, f5)
  f7(p1.jump.frame, f6)

  local function f8(p6, p7)
    local textButton2 = Instance.new("TextButton")
    textButton2.Size = UDim2.new(1, 0, 1, 0)
    textButton2.BackgroundTransparency = 1
    textButton2.Text = ""
    textButton2.Parent = p7.Parent.Parent

    textButton2.MouseButton1Click:Connect(function()
      p7.Text = "..."
      local connect4

      connect4 = userInputService.InputBegan:Connect(function(input3, p8)
        if p8 then
          return
        else
          local keyCode = input3.KeyCode

          if keyCode ~= Enum.KeyCode.Unknown then
            v4[p6] = keyCode
            p7.Text = keyCode.Name

            if connect4 then
              connect4:Disconnect()
            end
          end

          return
        end
      end)
    end)
  end

  f8("Fly", v5.Fly)
  f8("Speed", v5.Speed)
  f8("Jump", v5.Jump)

  userInputService.InputBegan:Connect(function(input4, p9)
    if p9 then
      return
    end

    if input4.KeyCode == v4.Fly then
      if v1 then
        f4()
      else
        f2()
      end
    elseif input4.KeyCode == v4.Speed then
      f5()
    elseif input4.KeyCode == v4.Jump then
      f6()
    end
  end)

  return { stopFly = f4, toggleSpeed = f5, toggleJump = f6 }
end
